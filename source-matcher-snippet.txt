      let t=String(r.result||'');
      /* PDF/DOCX read as raw bytes produce binary noise. Strip it so the
         matcher never receives junk it will silently fail on. */
      t=t.replace(/[^\x09\x0A\x0D\x20-\x7E\u0600-\u06FF]/g,' ')
         .replace(/\s{2,}/g,' ').trim();
      const letters=(t.match(/[A-Za-z\u0600-\u06FF]/g)||[]).length;
      /* Too little real text = unreadable format (scanned PDF, binary docx) */
      resolve(letters < 120 ? '' : t);
    };
    r.onerror=()=>resolve('');
    try{r.readAsText(file);}catch(e){resolve('');}
  });
}

function demoLists(cvText){
  const t=String(cvText||'').toLowerCase();
  const ASK=[{title:'Tell us your target roles',items:['Send your CV on WhatsApp','We read it manually','You get a real match']}];
  /* Too little readable text (scanned PDF / binary docx): never fake a match */
  if(t.length<25) return ASK;
  const scored=FIELD_MAP.map(function(f){
    var n=0; f.kw.forEach(function(k){ if(t.indexOf(k)>-1) n++; });
    return {f:f,score:n};
  }).filter(function(x){return x.score>0;}).sort(function(a,b){return b.score-a.score;});
  if(!scored.length) return ASK;
  /* Only show extra fields if they are a genuine signal, not a single
     stray keyword. A weak 2nd/3rd match is worse than showing one. */
  var top=scored[0].score;
  var keep=scored.filter(function(x,i){
    if(i===0) return true;
    return x.score>=2 && x.score>=top*0.6;
  });
  return keep.slice(0,3).map(function(x){return {title:x.f.title,items:x.f.items};});
}

}
